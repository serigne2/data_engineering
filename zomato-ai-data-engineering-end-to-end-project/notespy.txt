# ============================================================
# Load environment variables
# ============================================================

load_dotenv()


# ============================================================
# Configuration
# ============================================================

# ============================================================
# MinIO connection
# ============================================================

s3 = boto3.client(
    "s3",
    endpoint_url=MINIO_ENDPOINT,
    aws_access_key_id=MINIO_ACCESS_KEY,
    aws_secret_access_key=MINIO_SECRET_KEY,
    config=Config(signature_version="s3v4"),
    region_name="us-east-1",
)

print("✓ Connexion MinIO OK")


# ============================================================
# Snowflake connection
# ============================================================

conn = snowflake.connector.connect(

    warehouse="ZOMATO_WH",
    database="ZOMATO",
    schema="RAW",
)

cursor = conn.cursor()

print("✓ Connexion Snowflake OK")

# ============================================================
# Load one CSV
# ============================================================

def load_csv(minio_key, config, batch_size=10_000):
    table = config["table"]
    columns = config["columns"]

    print(f"\n→ Chargement de {minio_key}")
    print(f"  Destination : {table}")
    print(f"  Batch size  : {batch_size:,}")

    response = s3.get_object(Bucket=BUCKET, Key=minio_key)

    # Lecture correcte du CSV directement depuis le flux MinIO
    import io

    text_stream = io.TextIOWrapper(
        response["Body"],
        encoding="utf-8",
        newline=""
    )

    reader = csv.reader(text_stream)

    # Ignore le header
    next(reader)

    placeholders = ", ".join(["%s"] * len(columns))
    column_list = ", ".join(columns)

    sql = f"""
        INSERT INTO {table} ({column_list})
        VALUES ({placeholders})
    """

    batch = []
    row_count = 0

    for row in reader:

        # Conversion des chaînes vides en NULL
        row = [None if value == "" else value for value in row]

        batch.append(row)

        # Quand le batch atteint 10 000 lignes
        if len(batch) >= batch_size:
            cursor.executemany(sql, batch)

            row_count += len(batch)
            print(f"  ✓ {row_count:,} lignes chargées")

            batch = []

    # Charge le dernier batch s'il reste des lignes
    if batch:
        cursor.executemany(sql, batch)

        row_count += len(batch)
        print(f"  ✓ {row_count:,} lignes chargées")

    conn.commit()

    print(f"  ✓ Total : {row_count:,} lignes")
