def download_from_minio(minio_key):
    filename = os.path.basename(minio_key)
    local_path = os.path.join("/tmp", filename)

    print(f"\n→ Téléchargement de {minio_key}")
    print(f"  Destination locale : {local_path}")

    s3.download_file(
        BUCKET,
        minio_key,
        local_path
    )

    file_size_mb = os.path.getsize(local_path) / (1024 * 1024)

    print(f"  ✓ Téléchargement terminé : {file_size_mb:,.1f} MB")

    return local_path

import time


def bulk_load_to_snowflake(local_file, table, stage_path):
    """
    Charge un fichier local dans Snowflake via :
    PUT -> Internal Stage -> COPY INTO
    """

    print(f"\n→ Chargement BULK de {local_file}")
    print(f"  Destination : {table}")
    print(f"  Stage       : {stage_path}")

    start = time.time()

    # 1. Upload du fichier vers le stage interne Snowflake
    print("\n  [1/2] PUT vers Snowflake Internal Stage...")

    put_sql = f"""
        PUT file://{local_file}
        @{stage_path}
        AUTO_COMPRESS=TRUE
        PARALLEL=8
    """

    result = cursor.execute(put_sql).fetchall()

    put_time = time.time() - start

    print(f"  ✓ PUT terminé en {put_time / 60:.1f} min")

    for row in result:
        print(f"  {row}")

    # 2. COPY INTO Snowflake
    print("\n  [2/2] COPY INTO Snowflake...")

    copy_start = time.time()

    copy_sql = f"""
        COPY INTO {table}
        FROM @{stage_path}/
        FILE_FORMAT = (FORMAT_NAME = 'ZOMATO.RAW.CSV_FMT')
        ON_ERROR = 'ABORT_STATEMENT'
    """

    copy_result = cursor.execute(copy_sql).fetchall()

    copy_time = time.time() - copy_start

    print(f"  ✓ COPY terminé en {copy_time / 60:.1f} min")

    for row in copy_result:
        print(f"  {row}")

    conn.commit()

    total_time = time.time() - start

    print(f"\n✓ Chargement BULK terminé en {total_time / 60:.1f} min")

def load_csv(local_path, config, batch_size=10_000):
    table = config["table"]
    columns = config["columns"]

    print(f"\n→ Chargement de {local_path}")
    print(f"  Destination : {table}")
    print(f"  Batch size  : {batch_size:,}")

    with open(local_path, "r", encoding="utf-8", newline="") as file:
        reader = csv.reader(file)

        # Ignore le header
        next(reader)

        placeholders = ", ".join(["%s"] * len(columns))
        column_list = ", ".join(columns)

        sql = f"""
            INSERT INTO {table} ({column_list})
            VALUES ({placeholders})
        """

        batch = []
        row_count = 0

        for row in reader:

            row = [
                None if value == "" else value
                for value in row
            ]

            batch.append(row)

            if len(batch) >= batch_size:
                cursor.executemany(sql, batch)

                row_count += len(batch)

                print(f"  ✓ {row_count:,} lignes chargées")

                batch = []

        # Dernier batch
        if batch:
            cursor.executemany(sql, batch)

            row_count += len(batch)

            print(f"  ✓ {row_count:,} lignes chargées")

    conn.commit()

    print(f"  ✓ Total : {row_count:,} lignes")
